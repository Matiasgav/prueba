\section{Método}
Con $x$ la distancia al tope magnético ($x=0$: émbolo asentado), carrera $S$ y fracción recorrida $f=0{,}6$:
\begin{equation*}
W=\int_{(1-f)S}^{S}F(x)\,\mathrm{d}x,\qquad
\bar F=\frac{W}{fS},\qquad
\text{uniformidad}=\frac{F\big((1-f)S\big)}{F(S)},\qquad
E_c=\eta\,W=\tfrac12\,m_\mathrm{ef}\,v^2 .
\end{equation*}
$\eta$ (subida de corriente, fuerza contraelectromotriz, fricción, resorte) se mide en banco.

\textbf{Sobreexcitación.} Las curvas de 10\,\% ED son de régimen térmico, no un límite de fuerza: con un pulso
de 10--20~ms se puede aplicar más corriente, y en entrehierros grandes (circuito no saturado) la fuerza crece
aproximadamente con el cuadrado de la corriente. Es la palanca para que Ledex B12P o RD-A420 lleguen a 3~mJ,
pero queda \textbf{fuera de hoja} y hay que validarla en banco (temperatura, aislación, repetibilidad).

        carrera): fuera.
  \item \textbf{MSA «$>$1~M ciclos»} y \textbf{Delta EOL/100~k}: no verificables en las fuentes primarias.
\end{itemize}

\section{Decisión, ensayos y disparo}
\subsection{Decisión}
\textbf{Takaha CA0422 (CA04220240, 24~$\Omega$), a validar.} Es el único que cumple sección, energía y
uniformidad con la hoja en la mano. Tres preguntas a Takaha antes de diseñar: (1)~curva de 3 a 6~mm, (2)~masa
del émbolo y vida esperada, (3)~variante de \textbf{empuje} o varilla pasante. Si la vida no llega a millones de
ciclos en el ensayo propio, la alternativa de empuje es el \textbf{Ledex B12P con sobreexcitación} validada en banco.

\subsection{Ensayos}
\label{sec:ensayos}
\begin{enumerate}[leftmargin=1.4em]
  \item \textbf{Energía real ($\eta$):} punta de 1~g contra yunque instrumentado; velocidad con horquilla óptica
        doble; barrido de tensión y ancho de pulso, incluida sobreexcitación.
  \item \textbf{Repetibilidad:} 1000 disparos por condición; objetivo CV$(v)<2$\,\%.
  \item \textbf{Sensibilidad a la posición:} correr el yunque $\pm0{,}2$~mm.
  \item \textbf{Vida acelerada:} 3 unidades por modelo a 5--10~Hz hasta 5~M impactos; cada 500~k, velocidad,
        dispersión, juego y desgaste del émbolo.
  \item \textbf{Temperatura:} repetir 1 con bobina a 60--80~°C.
\end{enumerate}

\subsection{Retorno y disparo}
\begin{itemize}
  \item \textbf{Resorte de retorno externo:} ninguno de los candidatos pull/push sin enclavamiento lo trae. Un
        resorte de 0,1~N medio resta $\approx$0,24~mJ en 2,4~mm; dimensionarlo con eso en cuenta.
  \item \textbf{Driver de corriente controlada} y \textbf{corte antes del impacto} con apagado rápido (zéner/TVS): el
        impactador llega en vuelo libre y el ancho de pulso regula la energía.
  \item \textbf{Térmica:} 24~W$\times$15~ms $=0{,}36$~J por disparo; a 1~Hz son 0,36~W, debajo de 1,5~W continuos.
\end{itemize}

\subsection{Limitaciones}
\begin{itemize}
  \item \textbf{Curvas estáticas} de catálogo ($\pm10$\,\%): $W$ es cota superior.
  \item \textbf{Takaha de 3 a 4~mm extrapolado}; Transmotec cota inferior; MSS lectura aproximada.
  \item \textbf{Vida no publicada} para Takaha, Ledex B12 y MSS.
  \item \textbf{Mercado local:} no se pudo consultar MercadoLibre en forma automática; se buscó por buscador web.
  \item \textbf{Precios, stock y normativa} al 24/09/2026.
\end{itemize}
